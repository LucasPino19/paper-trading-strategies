// ⚡ Momentum 10d
// Timeframe: 1 hora (1H)  ← IMPORTANTE: usar gráfico de 1 hora
// Activo recomendado: NVDA (momentum explosivo, buenas tendencias)
// Backtest: +17.3% anual, 77% meses positivos
// Lógica: 70 barras de 1h ≈ 10 días de trading
//@version=6
strategy("⚡ Momentum 10d", overlay=true,
         initial_capital=10000,
         default_qty_type=strategy.percent_of_equity,
         default_qty_value=30,
         commission_type=strategy.commission.percent,
         commission_value=0.1,
         slippage=2)

// ── Parámetros ──────────────────────────────────────────────────────────────
mom_bars   = input.int(70,    "Barras momentum (70 = 10d en 1h)", group="Estrategia")
mom_thr    = input.float(4.0, "Umbral momentum (%)",              group="Estrategia") / 100
ema_period = input.int(50,    "EMA período",                      group="Estrategia")
stop_pct   = input.float(6.0, "Stop Loss (%)",                    group="Salidas") / 100
hard_tgt   = input.float(12.0,"Target (%)",                       group="Salidas") / 100
trail_trig = input.float(7.0, "Trail Trigger (%)",                group="Salidas") / 100
trail_pct  = input.float(4.0, "Trail desde pico (%)",             group="Salidas") / 100
time_stop  = input.int(70,    "Time Stop (barras 1h ≈ 10d)",      group="Salidas")

// ── Indicadores ─────────────────────────────────────────────────────────────
ema50    = ta.ema(close, ema_period)
// Momentum = retorno de los últimos N barras
momentum      = (close - close[mom_bars]) / close[mom_bars]
momentum_prev = (close[1] - close[mom_bars + 1]) / close[mom_bars + 1]

// Cruce hacia arriba del umbral: hoy >= 4%, ayer < 4%
cross_up = momentum >= mom_thr and momentum_prev < mom_thr

// ── Condiciones de entrada ───────────────────────────────────────────────────
long_signal = cross_up and close > ema50

// ── Gestión de posición ──────────────────────────────────────────────────────
var float peak_price = na
var int   entry_bar  = na
var bool  trail_on   = false

if long_signal and strategy.position_size == 0
    strategy.entry("Long", strategy.long)
    entry_bar  := bar_index
    peak_price := close
    trail_on   := false

if strategy.position_size > 0
    peak_price := math.max(peak_price, close)
    pnl        = (close - strategy.position_avg_price) / strategy.position_avg_price
    from_peak  = (close - peak_price) / peak_price
    bars_held  = bar_index - entry_bar

    if pnl >= trail_trig
        trail_on := true

    if pnl <= -stop_pct
        strategy.close("Long", comment="Stop -6%")
    else if pnl >= hard_tgt
        strategy.close("Long", comment="Target +12%")
    else if trail_on and from_peak <= -trail_pct
        strategy.close("Long", comment="Trail stop")
    else if bars_held >= time_stop
        strategy.close("Long", comment="Time stop")

// ── Plots ────────────────────────────────────────────────────────────────────
plot(ema50, "EMA50", color=color.orange, linewidth=2)
plotshape(long_signal and strategy.position_size == 0,
          "Momentum Entry", shape.labelup, location.belowbar,
          color.new(color.green, 0), size=size.small)

// Panel momentum en pane separado
mom_color = momentum >= mom_thr ? color.green : color.gray
plot(momentum * 100, "Momentum 10d (%)", color=mom_color, display=display.pane)
hline(mom_thr * 100, "Umbral 4%", color=color.new(color.green, 30), linestyle=hline.style_dashed)
hline(0, "0%", color=color.new(color.gray, 50))
