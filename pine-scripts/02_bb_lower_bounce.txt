// 🎯 BB Lower Bounce
// Timeframe: Daily (1D)
// Activo recomendado: SPY (ETF estable, revierte a la media)
// Backtest: +15.4% anual, 62% win rate, PF 2.29, DD -12%
//@version=5
strategy("🎯 BB Lower Bounce", overlay=true,
         initial_capital=10000,
         default_qty_type=strategy.percent_of_equity,
         default_qty_value=30,
         commission_type=strategy.commission.percent,
         commission_value=0.1,
         slippage=2)

// ── Parámetros ──────────────────────────────────────────────────────────────
bb_period  = input.int(20,    "BB Período",           group="Estrategia")
bb_std     = input.float(2.0, "BB Desviaciones",      group="Estrategia")
rsi_period = input.int(14,    "RSI Período",          group="Estrategia")
rsi_max    = input.float(45,  "RSI máximo entrada",   group="Estrategia")
ema_trend  = input.int(200,   "EMA Tendencia",        group="Estrategia")
stop_pct   = input.float(7.0, "Stop Loss (%)",        group="Salidas") / 100
hard_tgt   = input.float(15.0,"Target (%)",           group="Salidas") / 100
trail_trig = input.float(8.0, "Trail Trigger (%)",    group="Salidas") / 100
trail_pct  = input.float(4.0, "Trail desde pico (%)", group="Salidas") / 100
time_stop  = input.int(15,    "Time Stop (días)",     group="Salidas")

// ── Indicadores ─────────────────────────────────────────────────────────────
[bb_mid, bb_upper, bb_lower] = ta.bb(close, bb_period, bb_std)
rsi   = ta.rsi(close, rsi_period)
ema200 = ta.ema(close, ema_trend)

// ── Condiciones de entrada ───────────────────────────────────────────────────
// Ayer tocó o perforó la banda inferior
touched_bb = close[1] <= bb_lower[1]
// Hoy rebota
bounce     = close > close[1]
// RSI < 45 (oversold sin estar en caída libre)
rsi_ok     = rsi < rsi_max
// Tendencia alcista (sobre EMA200)
trend_up   = close > ema200

long_signal = touched_bb and bounce and rsi_ok and trend_up

// ── Gestión de posición ──────────────────────────────────────────────────────
var float peak_price = na
var int   entry_bar  = na
var bool  trail_on   = false

if long_signal and strategy.position_size == 0
    strategy.entry("Long", strategy.long)
    entry_bar  := bar_index
    peak_price := close
    trail_on   := false

if strategy.position_size > 0
    peak_price := math.max(peak_price, close)
    pnl        = (close - strategy.position_avg_price) / strategy.position_avg_price
    from_peak  = (close - peak_price) / peak_price
    days_held  = bar_index - entry_bar

    if pnl >= trail_trig
        trail_on := true

    if pnl <= -stop_pct
        strategy.close("Long", comment="Stop -7%")
    else if pnl >= hard_tgt
        strategy.close("Long", comment="Target +15%")
    else if trail_on and from_peak <= -trail_pct
        strategy.close("Long", comment="Trail stop")
    else if days_held >= time_stop
        strategy.close("Long", comment="Time stop " + str.tostring(days_held) + "d")

// ── Plots ────────────────────────────────────────────────────────────────────
p_upper = plot(bb_upper, "BB Upper", color=color.new(color.gray, 60))
p_lower = plot(bb_lower, "BB Lower", color=color.new(color.gray, 60))
fill(p_upper, p_lower, color=color.new(color.gray, 90))
plot(bb_mid,  "BB Mid",   color=color.gray, linewidth=1)
plot(ema200,  "EMA200",   color=color.blue, linewidth=2)
plotshape(long_signal and strategy.position_size == 0,
          "Entrada", shape.labelup, location.belowbar,
          color.new(color.green, 0), size=size.small)
bgcolor(touched_bb ? color.new(color.orange, 92) : na, title="BB tocada ayer")
