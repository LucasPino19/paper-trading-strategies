// 🧱 ICT Order Blocks + VWAP + EMA
// Timeframe: 1 hora (1H)  ← IMPORTANTE: usar gráfico de 1 hora
// Activo recomendado: SPY (Order Blocks bien respetados en ETF de índice)
// Backtest: +16.1% anual, 51% win rate, PF 1.72
// Lógica: vela bajista + 3 velas alcistas de impulso = OB alcista.
//         Precio vuelve a tocar la zona con VWAP y tendencia alineados.
//@version=6
strategy("🧱 ICT Order Blocks + VWAP", overlay=true,
         initial_capital=10000,
         default_qty_type=strategy.percent_of_equity,
         default_qty_value=30,
         commission_type=strategy.commission.percent,
         commission_value=0.1,
         slippage=2)

// ── Parámetros ──────────────────────────────────────────────────────────────
impulso    = input.int(3,     "Velas de impulso OB",           group="Estrategia")
ema_days   = input.int(20,    "EMA tendencia (días)",          group="Estrategia")
stop_pct   = input.float(7.0, "Stop Loss (%)",                 group="Salidas") / 100
hard_tgt   = input.float(20.0,"Target (%)",                    group="Salidas") / 100
trail_trig = input.float(10.0,"Trail Trigger (%)",             group="Salidas") / 100
trail_pct  = input.float(5.0, "Trail desde pico (%)",          group="Salidas") / 100
time_stop  = input.int(49,    "Time Stop (barras ≈ 7d)",       group="Salidas")

// ── Indicadores ─────────────────────────────────────────────────────────────
vwap_val  = ta.vwap(hlc3)

// Tendencia diaria proxy: EMA(20 días × 7 barras/día = 140 barras en 1h)
ema_trend = ta.ema(close, ema_days * 7)
trend_up  = close > ema_trend

// ── Detección OB Alcista ─────────────────────────────────────────────────────
// OB = vela bajista (índice `impulso+1`) seguida de `impulso` velas alcistas consecutivas
ob_candle_bearish = close[impulso + 1] < open[impulso + 1]

// Verificar impulso alcista posterior
bullish_run = true
if close[impulso] < open[impulso]
    bullish_run := false
if impulso >= 2 and close[impulso - 1] < open[impulso - 1]
    bullish_run := false
if impulso >= 3 and close[impulso - 2] < open[impulso - 2]
    bullish_run := false

ob_is_valid = ob_candle_bearish and bullish_run

// Zona del OB
ob_high = high[impulso + 1]
ob_low  = low[impulso + 1]
ob_size = ob_high - ob_low

// Precio toca la zona del OB en la barra actual
touch_ob = low <= ob_high and high >= ob_low and ob_size > 0

// ── Condiciones de entrada ───────────────────────────────────────────────────
above_vwap  = close > vwap_val
long_signal = ob_is_valid and touch_ob and above_vwap and trend_up

// ── Gestión de posición ──────────────────────────────────────────────────────
var float peak_price = na
var int   entry_bar  = na
var bool  trail_on   = false

if long_signal and strategy.position_size == 0
    strategy.entry("Long", strategy.long)
    entry_bar  := bar_index
    peak_price := close
    trail_on   := false

if strategy.position_size > 0
    peak_price := math.max(peak_price, close)
    pnl        = (close - strategy.position_avg_price) / strategy.position_avg_price
    from_peak  = (close - peak_price) / peak_price
    bars_held  = bar_index - entry_bar

    if pnl >= trail_trig
        trail_on := true

    if pnl <= -stop_pct
        strategy.close("Long", comment="Stop -7%")
    else if pnl >= hard_tgt
        strategy.close("Long", comment="Target +20%")
    else if trail_on and from_peak <= -trail_pct
        strategy.close("Long", comment="Trail stop")
    else if bars_held >= time_stop
        strategy.close("Long", comment="Time stop")

// ── Plots ────────────────────────────────────────────────────────────────────
plot(vwap_val,  "VWAP",      color=color.new(color.teal, 20), linewidth=2)
plot(ema_trend, "EMA 20d",   color=color.blue, linewidth=2)

// Marcar zona del OB
p_ob_high = plot(ob_is_valid ? ob_high : na, "OB High",
                 color=color.new(color.orange, 50), style=plot.style_linebr)
p_ob_low  = plot(ob_is_valid ? ob_low  : na, "OB Low",
                 color=color.new(color.orange, 50), style=plot.style_linebr)
fill(p_ob_high, p_ob_low,
     color=ob_is_valid ? color.new(color.orange, 88) : na,
     title="Zona OB")

plotshape(long_signal and strategy.position_size == 0,
          "OB Entry", shape.labelup, location.belowbar,
          color.new(color.green, 0), size=size.normal)
bgcolor(touch_ob and ob_is_valid ? color.new(color.blue, 92) : na,
        title="OB tocado")
