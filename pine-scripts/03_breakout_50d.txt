// 📈 Breakout 50d
// Timeframe: Daily (1D)
// Activo recomendado: QQQ (momentum fuerte, rompimientos limpios)
// Backtest: +14.5% anual, +89.8% total en 4.7 años
//@version=6
strategy("📈 Breakout 50d", overlay=true,
         initial_capital=10000,
         default_qty_type=strategy.percent_of_equity,
         default_qty_value=30,
         commission_type=strategy.commission.percent,
         commission_value=0.1,
         slippage=2)

// ── Parámetros ──────────────────────────────────────────────────────────────
lookback   = input.int(50,    "Lookback (días)",       group="Estrategia")
stop_pct   = input.float(6.0, "Stop Loss (%)",         group="Salidas") / 100
trail_trig = input.float(5.0, "Trail Trigger (%)",     group="Salidas") / 100
trail_pct  = input.float(8.0, "Trail desde pico (%)",  group="Salidas") / 100
time_stop  = input.int(30,    "Time Stop (días)",      group="Salidas")

// ── Indicadores ─────────────────────────────────────────────────────────────
// Máximo de los últimos N cierres ANTERIORES a la barra actual (sin look-ahead)
roll_high      = ta.highest(close[1], lookback)
roll_high_prev = ta.highest(close[2], lookback)

// Primera vez que cierra sobre el máximo: hoy sí, ayer no
first_break = close > roll_high and close[1] <= roll_high_prev

// ── Gestión de posición ──────────────────────────────────────────────────────
var float peak_price = na
var int   entry_bar  = na
var bool  trail_on   = false

if first_break and strategy.position_size == 0
    strategy.entry("Long", strategy.long)
    entry_bar  := bar_index
    peak_price := close
    trail_on   := false

if strategy.position_size > 0
    peak_price := math.max(peak_price, close)
    pnl        = (close - strategy.position_avg_price) / strategy.position_avg_price
    from_peak  = (close - peak_price) / peak_price
    days_held  = bar_index - entry_bar

    if pnl >= trail_trig
        trail_on := true

    if pnl <= -stop_pct
        strategy.close("Long", comment="Stop -6%")
    else if trail_on and from_peak <= -trail_pct
        strategy.close("Long", comment="Trail stop")
    else if days_held >= time_stop
        strategy.close("Long", comment="Time stop " + str.tostring(days_held) + "d")

// ── Plots ────────────────────────────────────────────────────────────────────
plot(roll_high, "50d High", color=color.new(color.purple, 50),
     linewidth=1, style=plot.style_stepline)
plotshape(first_break and strategy.position_size == 0,
          "Breakout", shape.labelup, location.belowbar,
          color.new(color.green, 0), size=size.normal)
bgcolor(first_break ? color.new(color.green, 88) : na, title="Breakout confirmado")
